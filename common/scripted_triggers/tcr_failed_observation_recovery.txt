# Recovery of the reported preenable observation stall only. The load-owned
# event and both ledgers must still describe the same untouched optional pass.
# This is deliberately independent of tcrv_identity_trigger, which rejects the
# saved failure and the pending load barrier that this recovery is inspecting.
tcrv_failed_observation_recovery_trigger = {
 trigger_if = { limit = {
  has_variable = tcrq_epoch has_variable = tcrc_lifecycle
  has_variable = tcrc_phase has_variable = tcrc_block_reason
  has_variable = tcrc_original_speed has_variable = tcrc_start_gold
  has_variable = tcrc_original_manual_pause has_variable = tcrc_manager_was_open
  exists = scope:manual exists = scope:manager_open
  has_variable = tcrc_life_year has_variable = tcrc_life_month has_variable = tcrc_life_day
  has_variable = tcr_active_generation
  has_variable = tcr_start_year has_variable = tcr_start_month has_variable = tcr_start_day
  has_variable = tcrv_owner has_variable = tcrv_generation has_variable = tcrv_lifecycle
  has_variable = tcrv_phase has_variable = tcrv_wallet has_variable = tcrv_moves
  has_variable = tcrv_member_count has_variable = tcrv_slot_count
  has_variable = tcrv_parent_accepted_count has_variable = tcrv_parent_recovery_class
  has_variable = tcrm_owner has_variable = tcrm_generation has_variable = tcrm_mode has_variable = tcrm_phase
  has_variable = tcrb_owner has_variable = tcrb_generation has_variable = tcrb_lifecycle has_variable = tcrb_phase
  has_variable = tcrb_settled_generation has_variable = tcrb_settled_lifecycle
 }
  tcrq_pending_event_trigger = yes tcro_current_player_owner_trigger = yes
  has_variable = tcrq_saved_failed has_variable = tcrq_event_seen
  has_variable = tcrc_lifecycle_active has_variable = tcr_failed
  has_variable = tca_enabled NOT = { has_variable = tcri_disable_requested }
  NOT = { has_variable = tcri_hold_reason } NOT = { has_variable = tcrc_abort_requested }
  NOT = { has_variable = tcrc_event_released }
  scope:epoch = var:tcrq_epoch scope:game_paused = yes scope:event_paused = yes
  scope:helpers_closed = yes scope:speed = var:tcrc_original_speed
  current_year = var:tcrc_life_year current_month = var:tcrc_life_month current_day = var:tcrc_life_day
  current_year = var:tcr_start_year current_month = var:tcr_start_month current_day = var:tcr_start_day
  gold = var:tcrc_start_gold gold = var:tcrv_wallet
  var:tcrc_phase = 90 var:tcrc_block_reason = 10
  var:tcrv_parent_recovery_class = 31
  has_variable = tcrv_active var:tcrv_phase = 10 var:tcrv_moves = 0
  var:tcrv_owner = this var:tcrv_generation = var:tcr_active_generation
  var:tcrv_lifecycle = var:tcrc_lifecycle var:tcrv_parent_accepted_count = 0
  tcrv_parent_ledger_identity_trigger = yes
  has_variable = tcrm_active var:tcrm_owner = this
  var:tcrm_generation = var:tcr_active_generation var:tcrm_mode = 2 var:tcrm_phase = 20
  has_variable = tcrb_settled var:tcrb_owner = this var:tcrb_phase = 70
  var:tcrb_generation = var:tcr_active_generation var:tcrb_lifecycle = var:tcrc_lifecycle
  var:tcrb_settled_generation = var:tcr_active_generation var:tcrb_settled_lifecycle = var:tcrc_lifecycle
  NOT = { has_variable = tcre_job_active } NOT = { has_variable = tcrp_job_active }
  NOT = { has_variable = tcrb_active } NOT = { has_variable = tcr_command_generation }
  NOT = { has_variable = tcrf_active } NOT = { has_variable = tcrf_pending }
  NOT = { has_variable = tcrf_receipt_pending }
  NOT = { has_variable = tcrv_job_payer } NOT = { has_variable = tcrv_job_source }
  NOT = { has_variable = tcrv_job_target } NOT = { has_variable = tcrv_expected_sequence }
  NOT = { has_variable = tcrv_parent_job_purpose } NOT = { has_variable = tcrv_hold }
  tcrv_no_pending_trigger = yes
 } trigger_else = { always = no }
}
