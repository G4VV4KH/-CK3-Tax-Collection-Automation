
tcrx_identity_trigger = {
 trigger_if = { limit = { has_variable = tcrx_owner has_variable = tcrx_lifecycle
  has_variable = tcrx_generation has_variable = tcrx_component has_variable = tcrx_phase
  has_variable = tcrx_target has_variable = tcrx_count
 }
  tcrs_lease_trigger = yes has_variable = tcrs_pending
  var:tcrx_owner = this var:tcrx_lifecycle = var:tcrs_lifecycle
  var:tcrx_generation = var:tcrs_generation var:tcrx_component = var:tcrs_sequence
 } trigger_else = { always = no }
}

tcrx_member_trigger = {
 trigger_if = { limit = { has_variable = tcrx_member_owner has_variable = tcrx_member_lifecycle
  has_variable = tcrx_member_generation has_variable = tcrx_member_component
  has_variable = tcrx_member_ordinal has_variable = tcrx_source has_variable = tcrx_target
 }
  var:tcrx_member_owner = root
  var:tcrx_member_lifecycle = root.var:tcrx_lifecycle
  var:tcrx_member_generation = root.var:tcrx_generation
  var:tcrx_member_component = root.var:tcrx_component
  var:tcrx_target = root.var:tcrx_target
  tcrs_original_payer_trigger = yes var:tcrx_source = var:tcrs_payer_original_slot
  save_temporary_scope_as = tcrx_member
  root = { is_target_in_variable_list = { name = tcrx_payers target = scope:tcrx_member } }
 } trigger_else = { always = no }
}
tcrx_expected_hook_trigger = {
 trigger_if = { limit = { has_variable = tcrx_active has_variable = tcrx_dispatched
  exists = scope:vassal exists = scope:tax_slot
 }
  tcrx_identity_trigger = yes var:tcrx_phase = 30
  save_temporary_scope_as = tcrx_hook_owner
  trigger_if = { limit = {
   is_target_in_variable_list = { name = tcrx_payers target = scope:vassal }
   scope:vassal = {
    has_variable = tcrx_member_owner has_variable = tcrx_member_lifecycle
    has_variable = tcrx_member_generation has_variable = tcrx_member_component
    has_variable = tcrx_source has_variable = tcrx_target
   }
  }
   scope:vassal = {
   var:tcrx_member_owner = scope:tcrx_hook_owner
   var:tcrx_member_lifecycle = scope:tcrx_hook_owner.var:tcrx_lifecycle
   var:tcrx_member_generation = scope:tcrx_hook_owner.var:tcrx_generation
   var:tcrx_member_component = scope:tcrx_hook_owner.var:tcrx_component
   }
   OR = { scope:tax_slot = var:tcrx_target scope:tax_slot = scope:vassal.var:tcrx_source }
  } trigger_else = { always = no }
 } trigger_else = { always = no }
}
tcrx_all_acked_trigger = {
 NOT = { any_in_list = { variable = tcrx_payers
  NOT = { trigger_if = { limit = { tcrx_member_trigger = yes
   has_variable = tcrx_assigned_ack has_variable = tcrx_removed_ack
   }
   var:tcrx_assigned_ack = root.var:tcrx_component
   var:tcrx_removed_ack = root.var:tcrx_component
   assigned_tax_slot = var:tcrx_target
  } trigger_else = { always = no } }
 } }
}
