# Advisory identity is useful only inside the current ordinary transaction.
# Recovery/disable/abort can retain data physically but cannot admit a recheck.
tcrv_advisory_identity_trigger = {
 trigger_if = { limit = {
  has_variable = tcrv_advisory_owner has_variable = tcrv_advisory_generation
  has_variable = tcrv_advisory_lifecycle has_variable = tcrv_generation
  has_variable = tcrv_lifecycle has_variable = tcr_active_generation
  has_variable = tcrc_lifecycle
 }
  tcrv_ordinary_trigger = yes
  var:tcrv_advisory_owner = this
  var:tcrv_advisory_generation = var:tcrv_generation
  var:tcrv_advisory_lifecycle = var:tcrv_lifecycle
  var:tcrv_generation = var:tcr_active_generation
  var:tcrv_lifecycle = var:tcrc_lifecycle
  NOT = { has_variable = tcrv_parent_recovery_readonly }
 } trigger_else = { always = no }
}
tcrv_recheck_identity_trigger = {

  tcrv_advisory_identity_trigger = yes
  is_target_in_variable_list = { name = tcrv_advisory_payers target = var:tcrv_recheck_payer }
  var:tcrv_recheck_payer = {

   tcrv_payer_unused_in_lifecycle_trigger = yes
  }
 } trigger_else = { always = no }
}


# Character scope; root is the optional transaction owner.
# FULL covers every not-exactly-used taxpayer. RECHECK covers exactly the
# selected advisory identity and rereads all its native monetary columns.
# Partial old move metadata stays covered by FULL, but cannot authorize a move.
tcrm_optional_measurement_required_trigger = {
 trigger_if = { limit = { root = { has_variable = tcrv_recheck_payer } }
  root = { tcrv_recheck_identity_trigger = yes }
  this = root.var:tcrv_recheck_payer
  tcrv_payer_unused_in_lifecycle_trigger = yes
 } trigger_else_if = { limit = { root = { has_variable = tcrv_lifecycle } }
  trigger_if = { limit = { has_variable = tcrv_move_owner has_variable = tcrv_move_lifecycle }
   NOT = { AND = { var:tcrv_move_owner = root var:tcrv_move_lifecycle = root.var:tcrv_lifecycle } }
  } trigger_else = { always = yes }
 } trigger_else = { always = no }
}

# Ruler scope. FULL has exact remaining-payer coverage; RECHECK has exact
# singleton coverage. Neither accepts cached monetary rows or a zero-row
# singleton. All-slot and all-fixed-member context checks remain unchanged.
tcrm_optional_measurement_coverage_trigger = {
 trigger_if = { limit = { has_variable = tcrv_lifecycle has_variable = tcrm_payer_count }
  trigger_if = { limit = { has_variable = tcrv_recheck_payer }
   tcrv_recheck_identity_trigger = yes
   var:tcrm_payer_count = 1
   is_target_in_variable_list = { name = tcrm_payers target = var:tcrv_recheck_payer }
   is_target_in_variable_list = { name = tcrm_fixed_payers target = var:tcrv_recheck_payer }
  }
  NOT = { any_in_list = { variable = tcrm_fixed_payers
   trigger_if = { limit = { tcrm_optional_measurement_required_trigger = yes }
    root = { NOT = { is_target_in_variable_list = { name = tcrm_payers target = prev } } }
   } trigger_else = {
    root = { is_target_in_variable_list = { name = tcrm_payers target = prev } }
   }
  } }
  NOT = { any_in_list = { variable = tcrm_payers
   OR = {
    NOT = { tcrm_optional_measurement_required_trigger = yes }
    root = { NOT = { is_target_in_variable_list = { name = tcrm_fixed_payers target = prev } } }
   }
  } }
 } trigger_else = { always = no }
}
