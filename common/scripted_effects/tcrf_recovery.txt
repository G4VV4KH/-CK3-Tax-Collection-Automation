# 39 unsupported/changed/ambiguous. No timer converts39 into success.

 set_variable = { name = tcrq_class value = 39 }
 if = { limit = {
trigger_if = { limit = {
has_variable = tcr_active_generation

var:tcrf_owner = this var:tcrf_lifecycle = var:tcrc_lifecycle

NOT = { has_variable = tcrf_failed } NOT = { has_variable = tcr_failed }

}
  if = { limit = { NOT = { has_variable = tcrf_pending } NOT = { has_variable = tcrf_receipt_pending }
   tcrf_recovery_no_command_trigger = yes
   OR = { var:tcrf_phase = 10 var:tcrf_phase = 12 var:tcrf_phase = 20 var:tcrf_phase = 71

  trigger_if = { limit = { has_variable = tcrf_ui_yield_lifecycle has_variable = tcrf_ui_yield_generation has_variable = tcrf_ui_yield_sequence has_variable = tcrf_parent_skip_optional_lifecycle }

   var:tcrf_ui_yield_sequence = var:tcrf_sequence var:tcrf_parent_skip_optional_lifecycle = var:tcrc_lifecycle
  } trigger_else = { always = no }
 }
}

  } set_variable = { name = tcrq_class value = 30 } }
  else_if = { limit = { has_variable = tcrf_pending has_variable = tcrf_authorized_sequence }
   if = { limit = { var:tcrf_authorized_sequence = var:tcrf_sequence }
    if = { limit = { var:tcrf_old = { is_tax_collector = no }
var:tcrf_new = { is_tax_collector_of = root tax_slot = root.var:tcrf_slot }

     OR = { tcrf_recovery_exact_command_trigger = yes tcrf_recovery_completed_target_trigger = yes }
    }
     tcrf_check_topology_effect = yes
     if = { limit = { NOT = { has_variable = tcrf_topology_invalid } } set_variable = { name = tcrq_class value = 32 } }
    }
    else_if = { limit = {
     var:tcrf_phase = 25 var:tcrf_fired_count = 1 var:tcrf_hired_count = 0
     var:tcrf_old = { is_tax_collector = no }
     var:tcrf_new = { is_alive = yes is_tax_collector = no }
     var:tcrf_slot = { has_tax_collector = no }
     tcrf_recovery_no_current_completion_trigger = yes
     OR = { tcrf_recovery_vacant_command_trigger = yes tcrf_recovery_no_command_trigger = yes }
    }
     tcrf_check_vacant_topology_effect = yes
     if = { limit = { NOT = { has_variable = tcrf_topology_invalid } } set_variable = { name = tcrq_class value = 31 } }
    }
   }
  }
 }

  if = { limit = { var:tcrf_phase = 73 }
   # Resume only existing no-command GUI quiescence after exact event reauth.
   set_variable = { name = tcrf_active value = yes }
   remove_variable = tcrc_manager_owned
  } else = {
   remove_variable = tcrf_active
   set_variable = { name = tcrf_phase value = 72 }

   # Actual target is recovery evidence, never a fabricated native fired/hired ACK.
   set_variable = { name = tcrf_recovered_target value = var:tcrq_epoch }
   set_variable = { name = tcrf_phase value = 30 }
  } else = {

   } else = { tcrf_fail_effect = yes }
  }
 }
