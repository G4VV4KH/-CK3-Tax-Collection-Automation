# Advisory payer order never authorizes a transfer. Native monetary rows
# are recreated by tcrm for the exact selected payer before each later job.
# Only identities/ranks survive; cached money cannot enter a receipt.
tcrv_clear_advisory_effect = {
 every_in_list = { variable = tcrv_advisory_payers
  limit = { exists = this }
  remove_variable = tcrv_advisory_rank
  remove_variable = tcrv_advisory_gain
 }
 clear_variable_list = tcrv_advisory_payers
 remove_variable = tcrv_advisory_owner
 remove_variable = tcrv_advisory_generation
 remove_variable = tcrv_advisory_lifecycle
 remove_variable = tcrv_advisory_continue
 remove_variable = tcrv_recheck_payer
 remove_variable = tcrv_advisory_next_rank
 remove_variable = tcrv_advisory_best_rank
 remove_variable = tcrv_advisory_build_payer
 remove_variable = tcrv_advisory_build_gain
}
# Character scope. Called only for a freshly priced, currently legal positive edge.
tcrv_record_advisory_gain_effect = {
 if = { limit = { scope:tcrv_owner = { NOT = { has_variable = tcrv_recheck_payer } } }
  if = { limit = { var:tcrv_gain > var:tcrv_advisory_gain }
   set_variable = { name = tcrv_advisory_gain value = var:tcrv_gain }
  }
 }
}
# Stable strict-max scan uses the provider's existing fixed iteration order for
# ties, exactly like the current best-edge selection. Store only ordinal ranks.
tcrv_build_advisory_effect = {
 set_variable = { name = tcrv_advisory_owner value = this }
 set_variable = { name = tcrv_advisory_generation value = var:tcrv_generation }
 set_variable = { name = tcrv_advisory_lifecycle value = var:tcrv_lifecycle }
 set_variable = { name = tcrv_advisory_next_rank value = 0 }
 while = {
  limit = { any_in_list = { variable = tcrm_payers
   has_variable = tcrv_advisory_gain var:tcrv_advisory_gain > 0
  } }
  remove_variable = tcrv_advisory_build_payer
  set_variable = { name = tcrv_advisory_build_gain value = 0 }
  every_in_list = { variable = tcrm_payers
   limit = { has_variable = tcrv_advisory_gain var:tcrv_advisory_gain > 0 }
   if = { limit = { var:tcrv_advisory_gain > scope:tcrv_owner.var:tcrv_advisory_build_gain }
    save_scope_as = tcrv_advisory_candidate
    scope:tcrv_owner = {
     set_variable = { name = tcrv_advisory_build_payer value = scope:tcrv_advisory_candidate }
     set_variable = { name = tcrv_advisory_build_gain value = scope:tcrv_advisory_candidate.var:tcrv_advisory_gain }
    }
   }
  }
  change_variable = { name = tcrv_advisory_next_rank add = 1 }
  add_to_variable_list = { name = tcrv_advisory_payers target = var:tcrv_advisory_build_payer }
  var:tcrv_advisory_build_payer = {
   set_variable = { name = tcrv_advisory_rank value = scope:tcrv_owner.var:tcrv_advisory_next_rank }
   remove_variable = tcrv_advisory_gain
  }
 }
 # Zero-gain scratch is not an advisory entry and is never retained.
 every_in_list = { variable = tcrm_payers remove_variable = tcrv_advisory_gain }
 remove_variable = tcrv_advisory_build_payer
 remove_variable = tcrv_advisory_build_gain
 remove_variable = tcrv_advisory_next_rank
}
# Ruler scope, phase5 with actual ledger and helper quiescence already checked.
# Invalid/stale queue identity is discarded before any provider is admitted.
tcrv_select_recheck_effect = {
 save_scope_as = tcrv_owner
 remove_variable = tcrv_recheck_payer
 remove_variable = tcrv_advisory_continue
 remove_variable = tcrv_advisory_best_rank
 if = { limit = { tcrv_advisory_identity_trigger = yes }
  every_in_list = { variable = tcrv_advisory_payers
   limit = {
    has_variable = tcrv_advisory_rank var:tcrv_advisory_rank > 0
    is_alive = yes exists = liege liege = scope:tcrv_owner
    exists = assigned_tax_slot tcrv_payer_unused_in_lifecycle_trigger = yes
   }
   if = { limit = { scope:tcrv_owner = { NOT = { has_variable = tcrv_recheck_payer } } }
    save_scope_as = tcrv_advisory_candidate
    scope:tcrv_owner = {
     set_variable = { name = tcrv_recheck_payer value = scope:tcrv_advisory_candidate }
     set_variable = { name = tcrv_advisory_best_rank value = scope:tcrv_advisory_candidate.var:tcrv_advisory_rank }
    }
   } else_if = { limit = { var:tcrv_advisory_rank < scope:tcrv_owner.var:tcrv_advisory_best_rank }
    save_scope_as = tcrv_advisory_candidate
    scope:tcrv_owner = {
     set_variable = { name = tcrv_recheck_payer value = scope:tcrv_advisory_candidate }
     set_variable = { name = tcrv_advisory_best_rank value = scope:tcrv_advisory_candidate.var:tcrv_advisory_rank }
    }
   }
  }
 }
 remove_variable = tcrv_advisory_best_rank
 if = { limit = { NOT = { has_variable = tcrv_recheck_payer } }
  tcrv_clear_advisory_effect = yes
 }
}

# Compare actual topology with the separate authorized ledger. Nothing below
# overwrites tcrc_original_payer_slot or tcrb_original_slot.
tcrv_check_ledger_effect = {
 save_scope_as = tcrv_owner
 remove_variable = tcrv_ledger_bad
 set_variable = { name = tcrv_actual_members value = 0 }
 set_variable = { name = tcrv_actual_slots value = 0 }
 every_tax_slot = {
  save_scope_as = tcrv_actual_slot
  scope:tcrv_owner = { change_variable = { name = tcrv_actual_slots add = 1 } }
  if = { limit = { NOT = { has_variable = tcrv_slot_owner has_variable = tcrv_slot_generation has_variable = tcrv_slot_lifecycle has_variable = tcrv_collector has_variable = tcrv_aptitude has_variable = tcrv_obligation } }
   scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } }
  } else_if = { limit = { has_tax_collector = no }
   scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } }
  } else_if = { limit = { OR = {
   var:tcrv_slot_owner != scope:tcrv_owner
   var:tcrv_slot_generation != scope:tcrv_owner.var:tcrv_generation
   var:tcrv_slot_lifecycle != scope:tcrv_owner.var:tcrv_lifecycle
   tax_collector != var:tcrv_collector
   tax_collector.tax_collector_aptitude_score != var:tcrv_aptitude
   NOT = { OR = { AND = { var:tcrv_obligation = 1 is_active_obligation = default_tax_collector } AND = { var:tcrv_obligation = 2 is_active_obligation = strict_taxation_tax_collector } AND = { var:tcrv_obligation = 3 is_active_obligation = iqta_special_rights_tax_collector } AND = { var:tcrv_obligation = 4 is_active_obligation = ghazi_special_rights_tax_collector } AND = { var:tcrv_obligation = 5 is_active_obligation = muqata_special_rights_tax_collector } AND = { var:tcrv_obligation = 6 is_active_obligation = jizya_special_rights_tax_collector } AND = { var:tcrv_obligation = 7 is_active_obligation = deqhan_special_rights_tax_collector } AND = { var:tcrv_obligation = 8 is_active_obligation = maguh_special_rights_tax_collector } } }
  } } scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } } }
  every_tax_slot_vassal = {
   scope:tcrv_owner = { change_variable = { name = tcrv_actual_members add = 1 } }
   if = { limit = { NOT = { has_variable = tcrv_member_owner has_variable = tcrv_member_generation has_variable = tcrv_member_lifecycle has_variable = tcrv_expected_slot } }
    scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } }
   } else_if = { limit = { OR = {
    var:tcrv_member_owner != scope:tcrv_owner
    var:tcrv_member_generation != scope:tcrv_owner.var:tcrv_generation
    var:tcrv_member_lifecycle != scope:tcrv_owner.var:tcrv_lifecycle
   } } scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } }
   } else_if = { limit = { assigned_tax_slot != var:tcrv_expected_slot }
    # Only the exact just-ACKed pending transfer may differ before acceptance.
    if = { limit = { scope:tcrv_owner = { has_variable = tcrv_job_payer has_variable = tcrv_job_target } }
     if = { limit = { NOT = { this = scope:tcrv_owner.var:tcrv_job_payer assigned_tax_slot = scope:tcrv_owner.var:tcrv_job_target scope:tcrv_owner = { tcrv_exact_job_completed_trigger = yes } } }
      scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } }
     }
    } else = { scope:tcrv_owner = { set_variable = { name = tcrv_ledger_bad value = yes } } }
   }
  }
 }
 if = { limit = { OR = {
  var:tcrv_actual_members != var:tcrv_member_count
  var:tcrv_actual_slots != var:tcrv_slot_count
  any_unassigned_taxpayers = { always = yes }
 } } set_variable = { name = tcrv_ledger_bad value = yes } }
}
