tcrc_preflight_effect = {
 remove_variable = tcrc_block_reason
 if = { limit = { NOT = { has_variable = tcr_collector_plan_ready } }
  set_variable = { name = tcrc_block_reason value = 1 }
 } else = {
  tcrc_build_attainable_plan_effect = yes
  save_scope_as = tcrc_owner
  every_in_list = { variable = tcrc_attainable_collectors
   save_scope_as = tcrc_candidate
   if = { limit = { OR = {
    NOT = { has_variable = tcrc_plan_slot }
    NOT = { has_variable = tcrc_plan_generation }
    var:tcrc_plan_generation != scope:tcrc_owner.var:tcr_active_generation
   } }
    scope:tcrc_owner = { set_variable = { name = tcrc_block_reason value = 4 } }
   } else_if = { limit = { NOT = { scope:tcrc_owner = {
      any_tax_slot = { this = scope:tcrc_candidate.var:tcrc_plan_slot }
   } } }
    scope:tcrc_owner = { set_variable = { name = tcrc_block_reason value = 4 } }
   } else_if = { limit = { NOT = { AND = {
     is_tax_collector_of = scope:tcrc_owner tax_slot = var:tcrc_plan_slot
   } } }
    if = { limit = { var:tcrc_plan_slot = { has_tax_collector = yes } }
     scope:tcrc_owner = { set_variable = { name = tcrc_block_reason value = 5 } }
    } else_if = { limit = { is_tax_collector = yes }
     scope:tcrc_owner = { set_variable = { name = tcrc_block_reason value = 6 } }
    }
   }
  }
  if = { limit = { NOT = { has_variable = tcrc_attainable_ready } }
   set_variable = { name = tcrc_block_reason value = 4 }
  }
 }
 if = { limit = { has_variable = tcrc_block_reason }
  tcrc_block_effect = yes
 } else = { set_variable = { name = tcrc_phase value = 30 } }
}

 clear_variable_list = tcrc_remaining_vacancies
 every_tax_slot = { remove_variable = tcrc_planned_collector }
 remove_variable = tcrc_attainable_ready
 remove_variable = tcrc_attainable_invalid
 remove_variable = tcrc_optional_deferred
 set_variable = { name = tcrc_plan_count value = 0 }
 set_variable = { name = tcrc_original_incumbent_count value = 0 }
 set_variable = { name = tcrc_remaining_vacancy_count value = 0 }
 set_variable = { name = tcrc_plan_slot_count value = var:tcrcp_slot_count }
 every_tax_collector = {
  save_scope_as = tcrc_incumbent
  if = { limit = { scope:tcrc_owner = { any_tax_slot = { this = scope:tcrc_incumbent.tax_slot } } }
   tax_slot = { save_scope_as = tcrc_target_slot }
   tcrc_add_attainable_pair_effect = yes
   scope:tcrc_owner = { change_variable = { name = tcrc_original_incumbent_count add = 1 } }
  } else = { scope:tcrc_owner = { set_variable = { name = tcrc_attainable_invalid value = yes } } }
 }
 if = { limit = { var:tcrc_original_incumbent_count = 0 }
  # Preserve the complete policy matching in the all-vacant stress case.
  every_in_list = { variable = tcap_selected_collectors
   var:tcr_plan_slot = { save_scope_as = tcrc_target_slot }
   tcrc_add_attainable_pair_effect = yes
  }
 } else = {
  # Bounded by actual slots. Strict comparisons preserve traversal ties.
  every_in_list = { variable = tcrcp_slots
   scope:tcrc_owner = {
    remove_variable = tcrc_best_slot
    remove_variable = tcrc_best_candidate
    every_in_list = { variable = tcrcp_slots
     if = { limit = { has_tax_collector = no NOT = { has_variable = tcrc_planned_collector } }
      save_scope_as = tcrc_slot_choice
      scope:tcrc_owner = {
       if = { limit = { NOT = { has_variable = tcrc_best_slot } }
        set_variable = { name = tcrc_best_slot value = scope:tcrc_slot_choice }
       } else_if = { limit = { scope:tcrc_slot_choice.var:tcrcp_policy_priority > var:tcrc_best_slot.var:tcrcp_policy_priority }
        set_variable = { name = tcrc_best_slot value = scope:tcrc_slot_choice }
       }
      }
     }
    }
    if = { limit = { has_variable = tcrc_best_slot }
     every_in_list = { variable = tcrcp_candidates
      save_scope_as = tcrc_candidate_choice
      if = { limit = { is_tax_collector = no tcrcp_candidate_eligible_trigger = yes
       NOT = { scope:tcrc_owner = { any_in_list = { variable = tcrc_attainable_collectors this = scope:tcrc_candidate_choice } } }
      }
       scope:tcrc_owner = {
        if = { limit = { NOT = { has_variable = tcrc_best_candidate } }
         set_variable = { name = tcrc_best_candidate value = scope:tcrc_candidate_choice }
        } else_if = { limit = { scope:tcrc_candidate_choice.var:tcrcp_aptitude > var:tcrc_best_candidate.var:tcrcp_aptitude }
         set_variable = { name = tcrc_best_candidate value = scope:tcrc_candidate_choice }
        }
       }
      }
     }
     if = { limit = { has_variable = tcrc_best_candidate }
      var:tcrc_best_slot = { save_scope_as = tcrc_target_slot }
      var:tcrc_best_candidate = { tcrc_add_attainable_pair_effect = yes }
     }
    }
   }
  }
 }
 every_in_list = { variable = tcap_selected_collectors
  save_scope_as = tcrc_candidate_choice
  if = { limit = { scope:tcrc_owner = { any_in_list = { variable = tcrc_attainable_collectors this = scope:tcrc_candidate_choice } } }
   if = { limit = { var:tcrc_plan_slot != var:tcr_plan_slot }
    scope:tcrc_owner = { set_variable = { name = tcrc_optional_deferred value = yes } }
   }
  } else = { scope:tcrc_owner = { set_variable = { name = tcrc_optional_deferred value = yes } } }
 }
 every_tax_slot = {
  if = { limit = { NOT = { has_variable = tcrc_planned_collector } }
   save_scope_as = tcrc_remaining_slot
   scope:tcrc_owner = {
    add_to_variable_list = { name = tcrc_remaining_vacancies target = scope:tcrc_remaining_slot }
    change_variable = { name = tcrc_remaining_vacancy_count add = 1 }
   }
  }
 }
 remove_variable = tcrc_best_candidate remove_variable = tcrc_best_slot
 if = { limit = { NOT = { has_variable = tcrc_attainable_invalid } var:tcrc_plan_count <= var:tcrc_plan_slot_count }
  set_variable = { name = tcrc_attainable_ready value = yes }
 }
