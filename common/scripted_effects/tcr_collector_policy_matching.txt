# Matrix contract: every payer has tcrcm_matrix_complete=yes only after provider
# has considered EVERY current slot. A missing tcrcm_policy_i is an illegal edge;
# it must never mean 'provider has not yet looked'. Capacity must match the
# proposed collector state, not merely the previous live state.
tcrcm_validate_matrix_effect = {
 remove_variable = tcrcm_matrix_valid
 remove_variable = tcrcm_matrix_missing
 every_in_list = {
  variable = tcrcm_payers
  if = { limit = { NOT = { has_variable = tcrcm_matrix_complete } }
   scope:tcrcm_owner = { set_variable = { name = tcrcm_matrix_missing value = yes } }
  }
 }
 if = { limit = {
  has_variable = tcrcm_snapshot_ready
  NOT = { has_variable = tcrcm_matrix_missing }
 } set_variable = { name = tcrcm_matrix_valid value = yes } }
}

# Longest augmenting path, compressed to slot vertices. An unassigned payer
# starts a path; an assigned payer transfers from its old slot. A path ending
# at a free slot can reassign several taxpayers, including cycles >2, in one
# uninterrupted script effect. No native tax state is changed while planning.
tcrcm_prepare_payer_relaxation_effect = {
 save_scope_as = tcrcm_payer
 scope:tcrcm_owner = {
  remove_variable = tcrcm_relax_ready
  set_variable = { name = tcrcm_base_policy value = 0 }
  set_variable = { name = tcrcm_base_keep value = 0 }
  if = { limit = { NOT = { scope:tcrcm_payer = { has_variable = tcrcm_plan_slot } } }
   set_variable = { name = tcrcm_relax_ready value = yes }
  }
  else_if = { limit = { scope:tcrcm_payer.var:tcrcm_plan_slot = { has_variable = tcrcm_dist_policy } }
   set_variable = { name = tcrcm_relax_ready value = yes }
   set_variable = { name = tcrcm_base_policy value = scope:tcrcm_payer.var:tcrcm_plan_slot.var:tcrcm_dist_policy }
   change_variable = { name = tcrcm_base_policy subtract = scope:tcrcm_payer.var:tcrcm_plan_policy }
   set_variable = { name = tcrcm_base_keep value = scope:tcrcm_payer.var:tcrcm_plan_slot.var:tcrcm_dist_keep }
   change_variable = { name = tcrcm_base_keep subtract = scope:tcrcm_payer.var:tcrcm_plan_keep }
  }
  if = { limit = { has_variable = tcrcm_relax_ready }
   tcrcm_relax_all_destinations_effect = yes
  }
 }
}

# INDEX is generated at build time; no dynamic identifier lookup is assumed.
tcrcm_relax_destination_effect = {
 if = { limit = {
  has_variable = tcrcm_slot_$INDEX$
  scope:tcrcm_payer = { has_variable = tcrcm_policy_$INDEX$ }
  OR = {
   NOT = { scope:tcrcm_payer = { has_variable = tcrcm_plan_slot } }
   scope:tcrcm_payer.var:tcrcm_plan_slot != var:tcrcm_slot_$INDEX$
  }
 }
  set_variable = { name = tcrcm_offer_policy value = var:tcrcm_base_policy }
  change_variable = { name = tcrcm_offer_policy add = scope:tcrcm_payer.var:tcrcm_policy_$INDEX$ }
  set_variable = { name = tcrcm_offer_keep value = var:tcrcm_base_keep }
  if = { limit = { scope:tcrcm_payer = { has_variable = tcrcm_native_slot }
    scope:tcrcm_payer.var:tcrcm_native_slot = var:tcrcm_slot_$INDEX$
   }
   change_variable = { name = tcrcm_offer_keep add = 1 }
  }
  var:tcrcm_slot_$INDEX$ = {
   if = { limit = { OR = {
    NOT = { has_variable = tcrcm_dist_policy }
    scope:tcrcm_owner.var:tcrcm_offer_policy > var:tcrcm_dist_policy
    AND = {
     scope:tcrcm_owner.var:tcrcm_offer_policy = var:tcrcm_dist_policy
     scope:tcrcm_owner.var:tcrcm_offer_keep > var:tcrcm_dist_keep
    }
   } }
    set_variable = { name = tcrcm_dist_policy value = scope:tcrcm_owner.var:tcrcm_offer_policy }
    set_variable = { name = tcrcm_dist_keep value = scope:tcrcm_owner.var:tcrcm_offer_keep }
    set_variable = { name = tcrcm_parent value = scope:tcrcm_payer }
   }
  }
 }
}

tcrcm_consider_endpoint_effect = {
 if = { limit = { has_variable = tcrcm_dist_policy var:tcrcm_used < var:tcrcm_capacity }
  save_scope_as = tcrcm_slot
  scope:tcrcm_owner = {
   if = { limit = { OR = {
    NOT = { has_variable = tcrcm_endpoint }
    scope:tcrcm_slot.var:tcrcm_dist_policy > var:tcrcm_endpoint_policy
    AND = { scope:tcrcm_slot.var:tcrcm_dist_policy = var:tcrcm_endpoint_policy
     scope:tcrcm_slot.var:tcrcm_dist_keep > var:tcrcm_endpoint_keep
    }
   } }
    set_variable = { name = tcrcm_endpoint value = scope:tcrcm_slot }
    set_variable = { name = tcrcm_endpoint_policy value = scope:tcrcm_slot.var:tcrcm_dist_policy }
    set_variable = { name = tcrcm_endpoint_keep value = scope:tcrcm_slot.var:tcrcm_dist_keep }
   }
  }
 }
}

tcrcm_augment_effect = {
 set_variable = { name = tcrcm_cursor_slot value = var:tcrcm_endpoint }
 set_variable = { name = tcrcm_backtrack_steps value = 0 }
 while = { limit = { has_variable = tcrcm_cursor_slot var:tcrcm_backtrack_steps < var:tcrcm_slot_count }
  change_variable = { name = tcrcm_backtrack_steps add = 1 }
  var:tcrcm_cursor_slot = { save_scope_as = tcrcm_slot var:tcrcm_parent = { save_scope_as = tcrcm_payer } }
  remove_variable = tcrcm_previous_slot
  if = { limit = { scope:tcrcm_payer = { has_variable = tcrcm_plan_slot } }
   set_variable = { name = tcrcm_previous_slot value = scope:tcrcm_payer.var:tcrcm_plan_slot }
   var:tcrcm_previous_slot = { change_variable = { name = tcrcm_used subtract = 1 } }
  }
  scope:tcrcm_payer = {
   set_variable = { name = tcrcm_plan_slot value = scope:tcrcm_slot }
   tcrcm_store_all_assignment_policy_effect = yes
   set_variable = { name = tcrcm_plan_keep value = 0 }
   if = { limit = { has_variable = tcrcm_native_slot var:tcrcm_native_slot = scope:tcrcm_slot }
    set_variable = { name = tcrcm_plan_keep value = 1 }
   }
  }
  scope:tcrcm_slot = { change_variable = { name = tcrcm_used add = 1 } }
  remove_variable = tcrcm_cursor_slot
  if = { limit = { has_variable = tcrcm_previous_slot }
   set_variable = { name = tcrcm_cursor_slot value = var:tcrcm_previous_slot }
  }
 }
 if = { limit = { has_variable = tcrcm_cursor_slot }
  set_variable = { name = tcrcm_plan_invalid value = yes }
 }
 change_variable = { name = tcrcm_planned_count add = 1 }
}

tcrcm_plan_taxpayers_effect = {
 save_scope_as = tcrcm_owner
 tcrcm_validate_matrix_effect = yes
 remove_variable = tcrcm_plan_complete
 remove_variable = tcrcm_plan_invalid
 if = { limit = { has_variable = tcrcm_matrix_valid }
  set_variable = { name = tcrcm_planned_count value = 0 }
  set_variable = { name = tcrcm_plan_loop value = yes }
  every_in_list = { variable = tcrcm_slots set_variable = { name = tcrcm_used value = 0 } }
  every_in_list = { variable = tcrcm_payers
   remove_variable = tcrcm_plan_slot remove_variable = tcrcm_plan_policy remove_variable = tcrcm_plan_keep
  }
  while = { limit = { has_variable = tcrcm_plan_loop NOT = { has_variable = tcrcm_plan_invalid } }
   every_in_list = { variable = tcrcm_slots
    remove_variable = tcrcm_dist_policy remove_variable = tcrcm_dist_keep remove_variable = tcrcm_parent
   }
   set_variable = { name = tcrcm_relax_round value = 0 }
   while = { limit = { var:tcrcm_relax_round < var:tcrcm_slot_count }
    every_in_list = { variable = tcrcm_payers tcrcm_prepare_payer_relaxation_effect = yes }
    change_variable = { name = tcrcm_relax_round add = 1 }
   }
   remove_variable = tcrcm_endpoint
   every_in_list = { variable = tcrcm_slots tcrcm_consider_endpoint_effect = yes }
   if = { limit = { var:tcrcm_planned_count < var:tcrcm_payer_count has_variable = tcrcm_endpoint OR = {
    var:tcrcm_endpoint_policy > 0
    AND = { var:tcrcm_endpoint_policy = 0 var:tcrcm_endpoint_keep >= 0 }
   } }
    tcrcm_augment_effect = yes
   }
   else = { remove_variable = tcrcm_plan_loop }
  }
  if = { limit = { NOT = { has_variable = tcrcm_plan_invalid } }
   set_variable = { name = tcrcm_plan_complete value = yes }
  }
 }
}
