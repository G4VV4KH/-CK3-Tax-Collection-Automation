if = { limit = { OR = { has_variable = tcrf_active has_variable = tcrf_pending has_variable = tcrf_receipt_pending } } tcrf_recovery_classify_effect = yes } else = {

  tcrv_parent_classify_recovery_effect = yes
 } else_if = { limit = {
  NOT = { has_variable = tcr_command_generation }
  NOT = { has_variable = tcre_job_active }
  NOT = { has_variable = tcrp_job_active }
 }
  set_variable = { name = tcrq_class value = 0 }
 } else_if = { limit = {
  has_variable = tcrq_command_schema has_variable = tcrq_source_kind
  has_variable = tcr_expected_kind has_variable = tcr_command_generation
  has_variable = tcr_active_generation
 }
  if = { limit = { var:tcrq_command_schema = 1 var:tcr_command_generation = var:tcr_active_generation }
   if = { limit = { var:tcr_expected_kind = 1 var:tcrq_source_kind = 1
    has_variable = tcre_job_active NOT = { has_variable = tcrp_job_active }
    has_variable = tcre_job_generation has_variable = tcre_sequence
   }
    if = { limit = { var:tcre_job_generation = var:tcr_active_generation }
     if = { limit = { exists = var:tcr_expected_collector exists = var:tcr_expected_slot }
      if = { limit = { any_tax_slot = { this = scope:tcrq_owner.var:tcr_expected_slot } }
       if = { limit = {
        var:tcr_expected_collector = { is_tax_collector_of = scope:tcrq_owner tax_slot = scope:tcrq_owner.var:tcr_expected_slot }
        var:tcr_expected_slot = { has_tax_collector = yes exists = tax_collector tax_collector = scope:tcrq_owner.var:tcr_expected_collector }
       } set_variable = { name = tcrq_class value = 1 }
       } else_if = { limit = { OR = {
        var:tcr_expected_collector = { is_tax_collector_of = scope:tcrq_owner tax_slot = scope:tcrq_owner.var:tcr_expected_slot }
        var:tcr_expected_slot = { has_tax_collector = no exists = tax_collector }
        var:tcr_expected_slot = { has_tax_collector = yes NOT = { exists = tax_collector } }
       } } set_variable = { name = tcrq_class value = 9 }
       } else_if = { limit = {
        var:tcr_expected_slot = { has_tax_collector = no }
        var:tcr_expected_collector = { is_alive = yes is_tax_collector = no }
       } set_variable = { name = tcrq_class value = 2 }
       } else = { set_variable = { name = tcrq_class value = 3 } }
      } else = { set_variable = { name = tcrq_class value = 3 } }
     } else = { set_variable = { name = tcrq_class value = 3 } }
    }
   } else_if = { limit = { var:tcr_expected_kind = 3 var:tcrq_source_kind = 0
    has_variable = tcrp_job_active NOT = { has_variable = tcre_job_active }
    has_variable = tcrp_job_generation has_variable = tcrp_sequence
    NOT = { has_variable = tcr_expected_source }
   }
    if = { limit = { var:tcrp_job_generation = var:tcr_active_generation }
     if = { limit = { exists = var:tcr_expected_payer exists = var:tcr_expected_slot }
      if = { limit = {
       any_tax_slot = { this = scope:tcrq_owner.var:tcr_expected_slot }
       var:tcr_expected_payer = { is_alive = yes exists = liege liege = scope:tcrq_owner }
      }
       if = { limit = {
        var:tcr_expected_payer = { exists = assigned_tax_slot assigned_tax_slot = scope:tcrq_owner.var:tcr_expected_slot }
        var:tcr_expected_slot = { any_tax_slot_vassal = { this = scope:tcrq_owner.var:tcr_expected_payer } }
        NOT = { any_tax_slot = {
         NOT = { this = scope:tcrq_owner.var:tcr_expected_slot }
         any_tax_slot_vassal = { this = scope:tcrq_owner.var:tcr_expected_payer }
        } }
       } set_variable = { name = tcrq_class value = 1 }
       } else_if = { limit = {
        var:tcr_expected_payer = { NOT = { exists = assigned_tax_slot } }
        NOT = { any_tax_slot = { any_tax_slot_vassal = { this = scope:tcrq_owner.var:tcr_expected_payer } } }
       } set_variable = { name = tcrq_class value = 2 }
       } else_if = { limit = {
        var:tcr_expected_payer = { exists = assigned_tax_slot NOT = { assigned_tax_slot = scope:tcrq_owner.var:tcr_expected_slot } }
        NOT = { var:tcr_expected_slot = { any_tax_slot_vassal = { this = scope:tcrq_owner.var:tcr_expected_payer } } }
       }
        # Read the new actual slot only after the existence guard above.
        if = { limit = {
         var:tcr_expected_payer = { assigned_tax_slot = { any_tax_slot_vassal = { this = scope:tcrq_owner.var:tcr_expected_payer } } }
         any_tax_slot = { this = scope:tcrq_owner.var:tcr_expected_payer.assigned_tax_slot }

       }
      } else = { set_variable = { name = tcrq_class value = 3 } }
     } else = { set_variable = { name = tcrq_class value = 3 } }
    }
   }
