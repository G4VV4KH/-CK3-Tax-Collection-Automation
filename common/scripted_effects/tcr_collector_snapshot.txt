tcrcp_add_candidate_effect = {
 if = {
  limit = { tcrcp_candidate_eligible_trigger = yes }
  save_scope_as = tcrcp_candidate
  scope:tcrcp_owner = {
   if = {
    limit = { NOT = { any_in_list = { variable = tcrcp_candidates this = scope:tcrcp_candidate } } }
    add_to_variable_list = { name = tcrcp_candidates target = scope:tcrcp_candidate }
    scope:tcrcp_candidate = { set_variable = { name = tcrcp_aptitude value = tax_collector_aptitude_score } }
   }
  }
 }
}


tcrcp_add_taxpayer_effect = {
 save_scope_as = tcrcp_payer
 scope:tcrcp_owner = {
  if = {
   limit = { NOT = { any_in_list = { variable = tcrcp_payers this = scope:tcrcp_payer } } }
   add_to_variable_list = { name = tcrcp_payers target = scope:tcrcp_payer }
   change_variable = { name = tcrcp_payer_count add = 1 }
   scope:tcrcp_payer = {
    if = { limit = { exists = assigned_tax_slot }
     set_variable = { name = tcrcp_native_slot value = assigned_tax_slot }
    }
   }
  }
 }
}


tcrcp_consider_candidate_effect = {
 if = { limit = { NOT = { has_variable = tcrcp_selected_rank } }
  save_scope_as = tcrcp_candidate
  scope:tcrcp_owner = {
   if = { limit = { NOT = { has_variable = tcrcp_best_candidate } }
    set_variable = { name = tcrcp_best_candidate value = scope:tcrcp_candidate }
    set_variable = { name = tcrcp_best_aptitude value = scope:tcrcp_candidate.var:tcrcp_aptitude }
   } else_if = { limit = { scope:tcrcp_candidate.var:tcrcp_aptitude > var:tcrcp_best_aptitude }
    set_variable = { name = tcrcp_best_candidate value = scope:tcrcp_candidate }
    set_variable = { name = tcrcp_best_aptitude value = scope:tcrcp_candidate.var:tcrcp_aptitude }
   }
  }
 }
}

tcrcp_select_top_collectors_effect = {
 set_variable = { name = tcrcp_selected_count value = 0 }
 # Iterating slots bounds selection by the actual jurisdiction count.
 every_in_list = {
  variable = tcrcp_slots
  scope:tcrcp_owner = {
   remove_variable = tcrcp_best_candidate
   remove_variable = tcrcp_best_aptitude
   every_in_list = { variable = tcrcp_candidates tcrcp_consider_candidate_effect = yes }
   if = {
    limit = { has_variable = tcrcp_best_candidate }
    add_to_variable_list = { name = tcap_selected_collectors target = var:tcrcp_best_candidate }
    var:tcrcp_best_candidate = {
     set_variable = { name = tcrcp_selected_rank value = scope:tcrcp_owner.var:tcrcp_selected_count }
    }
    change_variable = { name = tcrcp_selected_count add = 1 }
   }
  }
 }
 remove_variable = tcrcp_best_candidate
 remove_variable = tcrcp_best_aptitude
}


tcrcp_prepare_snapshot_effect = {
 save_scope_as = tcrcp_owner
 tcrcp_clear_snapshot_effect = yes
 set_variable = { name = tcrcp_slot_count value = 0 }
 set_variable = { name = tcrcp_payer_count value = 0 }
 every_tax_slot = {
  save_scope_as = tcrcp_slot
  set_variable = { name = tcrcp_index value = scope:tcrcp_owner.var:tcrcp_slot_count }
  set_variable = { name = tcrcp_capacity value = available_taxpayer_slots }
  every_tax_slot_vassal = {
   scope:tcrcp_slot = { change_variable = { name = tcrcp_capacity add = 1 } }
   tcrcp_add_taxpayer_effect = yes
  }
  scope:tcrcp_owner = {
   add_to_variable_list = { name = tcrcp_slots target = scope:tcrcp_slot }
   tcrcp_index_slot_effect = yes
   change_variable = { name = tcrcp_slot_count add = 1 }
  }
 }
 every_unassigned_taxpayers = { tcrcp_add_taxpayer_effect = yes }
 # Put incumbents first. Strict score comparison preserves them on equal score.
 every_tax_collector = { tcrcp_add_candidate_effect = yes }
 every_courtier = { tcrcp_add_candidate_effect = yes }
 every_vassal = { tcrcp_add_candidate_effect = yes }
 if = {
  limit = { var:tcrcp_slot_count <= 32 }
  tcrcp_select_top_collectors_effect = yes
  set_variable = { name = tcrcp_snapshot_ready value = yes }
 }
 else = { set_variable = { name = tcrcp_unsupported_slot_count value = yes } }
}
