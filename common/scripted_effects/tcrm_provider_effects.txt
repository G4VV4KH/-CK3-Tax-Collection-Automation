 scope:tcrm_owner = {
  clear_variable_list = tcrm_reachable_payers
  add_to_variable_list = { name = tcrm_reachable_payers target = scope:tcrm_unmatched }
  every_in_list = { variable = tcrm_slots remove_variable = tcrm_reachable }
  set_variable = { name = tcrm_reach_round value = 0 }
  while = { limit = { var:tcrm_reach_round <= var:tcrm_slot_count }
   every_in_list = { variable = tcrm_reachable_payers save_scope_as = tcrm_reach_payer
    scope:tcrm_owner = { tcrm_reach_destinations_effect = yes }
   }
   every_in_list = { variable = tcrm_payers
    limit = { has_variable = tcrm_plan_slot var:tcrm_plan_slot = { has_variable = tcrm_reachable } }
    save_scope_as = tcrm_reach_payer

     if = { limit = { NOT = { is_target_in_variable_list = { name = tcrm_reachable_payers target = scope:tcrm_reach_payer } } }
      add_to_variable_list = { name = tcrm_reachable_payers target = scope:tcrm_reach_payer }
     }
    }
   }
   change_variable = { name = tcrm_reach_round add = 1 }
  }
  if = { limit = { any_in_list = { variable = tcrm_slots has_variable = tcrm_reachable var:tcrm_used < var:tcrm_capacity } }
   scope:tcrm_unmatched = { set_variable = { name = tcrm_unmatched_reason value = 3 } }
  } else_if = { limit = { scope:tcrm_unmatched.var:tcrm_legal_edges = 0
    any_in_list = { variable = tcrm_slots var:tcrm_capacity > 0 }
   }
   scope:tcrm_unmatched = { set_variable = { name = tcrm_unmatched_reason value = 1 } }
  } else = { scope:tcrm_unmatched = { set_variable = { name = tcrm_unmatched_reason value = 2 } } }
 }
}

# Full matcher output is read-only scratch. This stage does not invoke tcrp.
tcrm_build_apply_order_effect = {
 clear_variable_list = tcrm_apply_order
 remove_variable = tcrm_apply_ready
 remove_variable = tcrm_apply_cycle_blocked
 remove_variable = tcrm_plan_drops_original
 set_variable = { name = tcrm_output_gold value = 0 }
 set_variable = { name = tcrm_original_gold value = 0 }
 set_variable = { name = tcrm_pending_moves value = 0 }
 every_in_list = { variable = tcrm_slots
  set_variable = { name = tcrm_simulated_used value = var:tcrm_native_used }
 }
 every_in_list = { variable = tcrm_payers
  remove_variable = tcrm_ordered
  remove_variable = tcrm_unmatched_reason
 }
 every_in_list = { variable = tcrm_payers
  save_scope_as = tcrm_payer
  if = { limit = { has_variable = tcrm_source_gold }
   scope:tcrm_owner = { change_variable = { name = tcrm_original_gold add = scope:tcrm_payer.var:tcrm_source_gold } }
  }
  if = { limit = { has_variable = tcrm_plan_slot }
   scope:tcrm_owner = { change_variable = { name = tcrm_output_gold add = scope:tcrm_payer.var:tcrm_plan_gold } }
   if = { limit = { OR = { NOT = { has_variable = tcrm_native_slot } var:tcrm_plan_slot != var:tcrm_native_slot } }
    scope:tcrm_owner = { change_variable = { name = tcrm_pending_moves add = 1 } }
   } else = { set_variable = { name = tcrm_ordered value = yes } }
  } else = {
   if = { limit = { has_variable = tcrm_native_slot }
    scope:tcrm_owner = { set_variable = { name = tcrm_plan_drops_original value = yes } }
   }
   tcrm_classify_unmatched_effect = yes
  }
 }
 set_variable = { name = tcrm_order_progress value = yes }
 while = { limit = { has_variable = tcrm_order_progress var:tcrm_pending_moves > 0 }
  remove_variable = tcrm_order_progress
  every_in_list = { variable = tcrm_payers
   limit = { has_variable = tcrm_plan_slot NOT = { has_variable = tcrm_ordered }
    var:tcrm_plan_slot = { var:tcrm_simulated_used < var:tcrm_native_capacity }
   }
   save_scope_as = tcrm_payer
   if = { limit = { has_variable = tcrm_native_slot }
    var:tcrm_native_slot = { change_variable = { name = tcrm_simulated_used subtract = 1 } }
   }
   var:tcrm_plan_slot = { change_variable = { name = tcrm_simulated_used add = 1 } }
   set_variable = { name = tcrm_ordered value = yes }
   scope:tcrm_owner = {
    add_to_variable_list = { name = tcrm_apply_order target = scope:tcrm_payer }
    change_variable = { name = tcrm_pending_moves subtract = 1 }
    set_variable = { name = tcrm_order_progress value = yes }
   }
  }
 }
 if = { limit = { var:tcrm_pending_moves > 0 }
  set_variable = { name = tcrm_apply_cycle_blocked value = yes }
 }
 if = { limit = {
  tcrm_context_valid_trigger = yes
  has_variable = tcrm_plan_complete
  NOT = { has_variable = tcrm_plan_invalid }
  NOT = { has_variable = tcrm_plan_drops_original }
  NOT = { has_variable = tcrm_apply_cycle_blocked }
  var:tcrm_output_gold >= var:tcrm_original_gold
 }
  set_variable = { name = tcrm_apply_ready value = yes }
 } else = {
  # Do not apply a low-income prefix while leaving its compensating cycle undone.
  clear_variable_list = tcrm_apply_order
 }
}

# No native command or newly sampled money is produced in this loop. It can
# consume only an already-measured source column or a proven full destination.
# An actual preview, a new source model, or a new payer is a hard yield point.
tcrm_drain_script_edges_effect = {
 if = { limit = {
  NOT = { has_variable = tcrm_drain_running }
  tcrm_script_drain_live_trigger = yes
 }
  set_variable = { name = tcrm_drain_running value = yes }
  set_variable = { name = tcrm_drain_steps value = 0 }
  set_variable = { name = tcrm_drain_expected_sequence value = var:tcrm_sequence }
  var:tcrm_current_payer = { save_scope_as = tcrm_drain_payer }
  while = {
   limit = {
    trigger_if = { limit = { tcrm_script_drain_live_trigger = yes }
     var:tcrm_drain_steps < 32
     var:tcrm_current_payer = scope:tcrm_drain_payer
     var:tcrm_sequence = var:tcrm_drain_expected_sequence
     OR = {
      tcrm_script_source_edge_trigger = yes
      tcrm_capacity_edge_unavailable_trigger = yes
     }
    } trigger_else = { always = no }
   }
   if = { limit = { tcrm_script_source_edge_trigger = yes }
    tcrm_write_script_source_edge_effect = yes
   } else = {
    tcrm_write_script_capacity_edge_effect = yes
   }
   change_variable = { name = tcrm_drain_steps add = 1 }
   change_variable = { name = tcrm_drain_expected_sequence add = 1 }
   tcrm_edge_finished_effect = yes
  }
  remove_variable = tcrm_drain_running
  remove_variable = tcrm_drain_steps
  remove_variable = tcrm_drain_expected_sequence
 }
}
