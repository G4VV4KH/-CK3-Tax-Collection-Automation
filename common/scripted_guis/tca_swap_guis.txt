  var:tca_swap_tx_b = { is_alive = yes liege = root NOT = { exists = assigned_tax_slot } can_assign_to_tax_slot = root.var:tca_swap_tx_t } } }

tca_swap_tx_finish = {
 scope = character
 is_valid = { tca_swap_tx_can_finish_trigger = yes }
 effect = { if = { limit = { tca_swap_tx_can_finish_trigger = yes }

  # Called only after the GUI restores native autoassign and pause settings.
  if = { limit = {
   var:tca_swap_tx_a = { assigned_tax_slot = root.var:tca_swap_tx_t }
   var:tca_swap_tx_b = { assigned_tax_slot = root.var:tca_swap_tx_s }
  }
   # A profitable exchange changes which pairs are profitable. Restart the
   # pair cursor without redoing recruitment or the direct-assignment pass.
   set_variable = { name = tca_swap_i value = 1 }
   set_variable = { name = tca_swap_j value = 2 }
  }
  else = {
   change_variable = { name = tca_swap_j add = 1 }
   if = { limit = { var:tca_swap_j > var:tca_swap_slot_count }
    change_variable = { name = tca_swap_i add = 1 }
    set_variable = { name = tca_swap_j value = var:tca_swap_i }
    change_variable = { name = tca_swap_j add = 1 }
   }
  }
  tca_swap_tx_clear_effect = yes
  if = { limit = { tca_enabled_trigger = yes } tca_swap_load_pair_effect = yes }

 } }
